Usage Examples

1. Basic Usage

// Check if extension is available
if (window.EstEID_PD) {
  // Read personal data
  const result = await EstEID_PD.readPersonalData();
  
  if (result.ok) {
    console.log('Name:', result.data.givenName, result.data.surname);
    console.log('Personal Code:', result.data.personalCode);
  } else {
    console.error('Error:', result.error);
  }
}

2. With Error Handling

try {
  const result = await EstEID_PD.readPersonalData({
    useCache: false,  // Force fresh read
    timeout: 30000    // 30 second timeout
  });
  
  if (result.ok) {
    // Use the data
    document.getElementById('name').textContent = 
      result.data.givenName + ' ' + result.data.surname;
  }
} catch (error) {
  alert('Failed to read ID card: ' + error.message);
}

3. Check Availability

// Check if extension is loaded and ready
const status = await EstEID_PD.checkStatus();

if (status.ok && status.available) {
  console.log('Extension is ready!');
  // Show "Read ID Card" button
} else {
  console.warn('Extension not available:', status.error);
  // Show installation instructions
}
